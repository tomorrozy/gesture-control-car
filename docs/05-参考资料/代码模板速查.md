# 代码模板（填空式）

> [!tip] 怎么用
> 每个模板都是**能跑的最小骨架**，只有「★」标记的地方需要你按自己的板子填。
> **用完记得删掉没用到的部分** —— 模板的价值是「不漏关键步骤」，不是「代码越多越好」。
>
> **江科大风格**（结构体初始化 + 中文注释），Keil MDK + 标准外设库，**勾 MicroLIB**。

## 0 通用约定（★ 先看这个）

| 约定 | 内容 |
|---|---|
| ★ **代码文件不加 YAML frontmatter、不加 BOM** | 否则 Keil 编译可能出问题 |
| ★ **中断里绝不死等、绝不做 I2C** | 短、纯、不碰共享资源 |
| ★ **串口发送不放中断** | 阻塞会拖垮控制周期 |
| ★ **中断与主循环共享的变量加 `volatile`** | 否则 `-O2` 下会被优化掉 |
| ★ **多字节共享变量要保护** | `float` 读写不是原子的 |
| ★ **每个 `Init()` 里写注释：配的是哪个寄存器、配错会怎样** | 这是防止「忘了初始化」的最好办法 |

## 1 串口（阻塞发送 + RXNE 中断接收）

```c
/* ---------- Serial.h ---------- */
void Serial_Init(void);                      /* ★ 默认 115200 */
void Serial_SendByte(uint8_t Byte);
void Serial_SendString(char *String);
void Serial_SendArray(uint8_t *Array, uint16_t Len);

extern volatile uint8_t Serial_RxData;
extern volatile uint8_t Serial_RxFlag;

/* ---------- Serial.c ---------- */
volatile uint8_t Serial_RxData;
volatile uint8_t Serial_RxFlag;

void Serial_Init(void)
{
    /* ① 时钟：★ GPIOA 和 USART1 是两个独立外设，都必须开 */
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA,  ENABLE);
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_USART1, ENABLE);

    /* ② 引脚：★ TX 必须复用推挽；RX 上拉输入 */
    GPIO_InitTypeDef GPIO_InitStructure;
    GPIO_InitStructure.GPIO_Mode  = GPIO_Mode_AF_PP;
    GPIO_InitStructure.GPIO_Pin   = GPIO_Pin_9;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);

    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IPU;
    GPIO_InitStructure.GPIO_Pin  = GPIO_Pin_10;
    GPIO_Init(GPIOA, &GPIO_InitStructure);

    /* ③ USART：8N1、无流控、收发都开 */
    USART_InitTypeDef USART_InitStructure;
    USART_InitStructure.USART_BaudRate            = 115200;   /* ★ 改这里 */
    USART_InitStructure.USART_WordLength          = USART_WordLength_8b;
    USART_InitStructure.USART_StopBits            = USART_StopBits_1;
    USART_InitStructure.USART_Parity              = USART_Parity_No;
    USART_InitStructure.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStructure.USART_Mode                = USART_Mode_Tx | USART_Mode_Rx;
    USART_Init(USART1, &USART_InitStructure);

    /* ④ 中断：★ 外设级 + 内核级，两层都要 */
    USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);

    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
    NVIC_InitTypeDef NVIC_InitStructure;
    NVIC_InitStructure.NVIC_IRQChannel                   = USART1_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority        = 1;
    NVIC_InitStructure.NVIC_IRQChannelCmd                = ENABLE;
    NVIC_Init(&NVIC_InitStructure);

    USART_Cmd(USART1, ENABLE);                /* ★ 忘了这行 = 全部白配 */
}

void Serial_SendByte(uint8_t Byte)
{
    USART_SendData(USART1, Byte);
    while (USART_GetFlagStatus(USART1, USART_FLAG_TXE) == RESET);
}

void Serial_SendString(char *String)
{
    uint8_t i;
    for (i = 0; String[i] != '\0'; i++) { Serial_SendByte(String[i]); }
}

void USART1_IRQHandler(void)
{
    if (USART_GetITStatus(USART1, USART_IT_RXNE) == SET)
    {
        Serial_RxData = USART_ReceiveData(USART1);
        Serial_RxFlag = 1;
        USART_ClearITPendingBit(USART1, USART_IT_RXNE);
    }
}
```

> **★ 自测清单**：`USART_Cmd` 写了吗？TX 是 `AF_PP` 吗？NVIC 配了吗？三个都对了才可能通。

## 2 定时中断（10ms 控制周期，服务多个模块）

```c
/* ---------- Timer.h ---------- */
void Timer_Init(void);                       /* ★ TIM1，10ms */
uint32_t Timer_GetMs(void);
void Timer_TickHook(void);                   /* ★ 必须由 main.c 实现！ */

/* ---------- Timer.c ---------- */
static volatile uint32_t s_ms = 0;

void Timer_Init(void)
{
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_TIM1, ENABLE);   /* ★ TIM1 在 APB2 */

    TIM_InternalClockConfig(TIM1);

    TIM_TimeBaseInitTypeDef TIM_TimeBaseInitStructure;
    TIM_TimeBaseInitStructure.TIM_ClockDivision = TIM_CKD_DIV1;
    TIM_TimeBaseInitStructure.TIM_CounterMode   = TIM_CounterMode_Up;
    TIM_TimeBaseInitStructure.TIM_Prescaler     = 7200 - 1;    /* ★ 真实分频 = 值 + 1 → 10kHz */
    TIM_TimeBaseInitStructure.TIM_Period        = 100  - 1;    /* ★ 真实周期 = 值 + 1 → 10ms */
    TIM_TimeBaseInitStructure.TIM_RepetitionCounter = 0;
    TIM_TimeBaseInit(TIM1, &TIM_TimeBaseInitStructure);

    TIM_ClearFlag(TIM1, TIM_FLAG_Update);
    TIM_ITConfig(TIM1, TIM_IT_Update, ENABLE);

    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);
    NVIC_InitTypeDef NVIC_InitStructure;
    NVIC_InitStructure.NVIC_IRQChannel                   = TIM1_UP_IRQn;   /* ★ 注意是 UP */
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 0;              /* ★ 控制最高优先级 */
    NVIC_InitStructure.NVIC_IRQChannelSubPriority        = 0;
    NVIC_InitStructure.NVIC_IRQChannelCmd                = ENABLE;
    NVIC_Init(&NVIC_InitStructure);

    TIM_Cmd(TIM1, ENABLE);
}

/* ★★ 函数名必须是 TIM1_UP_IRQHandler！
      写成 TIM1_IRQHandler 会「能编译、定时器在跑、但永远不进中断」 */
void TIM1_UP_IRQHandler(void)
{
    if (TIM_GetITStatus(TIM1, TIM_IT_Update) == SET)
    {
        TIM_ClearITPendingBit(TIM1, TIM_IT_Update);
        s_ms += 10;
        Timer_TickHook();                     /* ★ 具体业务由 main.c 实现 */
    }
}

uint32_t Timer_GetMs(void) { return s_ms; }
```

> ⚠️ **不要用 SysTick 做时基** —— 江科大 `Delay_us()` 也用它，而且会把 `TICKINT` 关掉
> （表现为「一调 `Delay_us`，10ms 中断就没了」）。

## 3 PWM 输出（20kHz，ARR=99 让 CCR 直接当百分数）

```c
/* ---------- Motor.c ---------- */
#define PWM_PSC   36        /* ★ 真实分频 36 → 2MHz */
#define PWM_ARR   99        /* ★ 真实周期数 100 → 2MHz/100 = 20kHz */

void Motor_Init(void)
{
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM2, ENABLE);    /* ★ TIM2 在 APB1 */
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);

    /* PA2 = TIM2_CH3，PA3 = TIM2_CH4 → ★ 复用推挽 */
    GPIO_InitTypeDef GPIO_InitStructure;
    GPIO_InitStructure.GPIO_Mode  = GPIO_Mode_AF_PP;
    GPIO_InitStructure.GPIO_Pin   = GPIO_Pin_2 | GPIO_Pin_3;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);

    TIM_InternalClockConfig(TIM2);

    TIM_TimeBaseInitTypeDef TIM_TimeBaseInitStructure;
    TIM_TimeBaseInitStructure.TIM_ClockDivision = TIM_CKD_DIV1;
    TIM_TimeBaseInitStructure.TIM_CounterMode   = TIM_CounterMode_Up;
    TIM_TimeBaseInitStructure.TIM_Prescaler     = PWM_PSC - 1;
    TIM_TimeBaseInitStructure.TIM.Period        = PWM_ARR;      /* 见下注 */
    TIM_TimeBaseInitStructure.TIM_Period        = PWM_ARR;
    TIM_TimeBaseInitStructure.TIM_RepetitionCounter = 0;
    TIM_TimeBaseInit(TIM2, &TIM_TimeBaseInitStructure);

    TIM_OCInitTypeDef TIM_OCInitStructure;
    TIM_OCStructInit(&TIM_OCInitStructure);
    TIM_OCInitStructure.TIM_OCMode      = TIM_OCMode_PWM1;
    TIM_OCInitStructure.TIM_OutputState = TIM_OutputState_Enable;   /* ★ 使能输出，引脚交给定时器 */
    TIM_OCInitStructure.TIM_Pulse       = 0;                        /* 初始 0% */
    TIM_OCInitStructure.TIM_OCPolarity  = TIM_OCPolarity_High;
    TIM_OC3Init(TIM2, &TIM_OCInitStructure);      /* PA2 */
    TIM_OC4Init(TIM2, &TIM_OCInitStructure);      /* PA3 */

    TIM_OC3PreloadConfig(TIM2, TIM_OCPreload_Enable);
    TIM_OC4PreloadConfig(TIM2, TIM_OCPreload_Enable);
    TIM_ARRPreloadConfig(TIM2, ENABLE);

    TIM_Cmd(TIM2, ENABLE);
}

/* ★ 防御性编程：函数内部也限一次，不信任调用者 */
void Motor_SetSpeed(uint8_t ch, uint8_t pwm)
{
    if (pwm > 100) { pwm = 100; }                 /* ★ 内部再限一次 */
    if (ch == 0) { TIM_SetCompare3(TIM2, pwm); }
    else         { TIM_SetCompare4(TIM2, pwm); }
}
```

> **★ 注意 `TIM_Period = PWM_ARR`（不是 `PWM_ARR - 1`）**：
> 这里我故意写成 `ARR = 99` 让「一个周期 = 100 个数」，**`CCR` 直接就是百分数**，代码更干净。
> 代价：分辨率只有 1%（够用，见原理讲解第 18 篇）。
>
> **★ 换向时先清零 PWM**（否则电流冲击接近堵转）：
> ```c
> Motor_SetSpeed(0, 0);  Delay_us(10);  /* ★ 再改方向脚 */  Motor_SetSpeed(0, target);
> ```

## 4 编码器接口（四倍频 + 每周期清零）

```c
/* ---------- Encoder.h ---------- */
#define ENCODER_CNT_PER_REV   (4.0f * 13.0f * 30.0f)      /* ★ 四倍频 x PPR x 减速比 */
#define ENCODER_DIR_L         1.0f                        /* ★ 同号性：反了就改 -1 */
#define ENCODER_DIR_R         1.0f

void Encoder_Init(void);
int16_t Encoder_Get(void);          /* ★ 读增量并清零（避免 CNT 回绕） */

/* ---------- Encoder.c ---------- */
void Encoder_Init(void)
{
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM3, ENABLE);
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);

    /* PA6 = TIM3_CH1 = A 相，PA7 = TIM3_CH2 = B 相 */
    GPIO_InitTypeDef GPIO_InitStructure;
    GPIO_InitStructure.GPIO_Mode  = GPIO_Mode_IPU;        /* ★ 上拉输入最稳 */
    GPIO_InitStructure.GPIO_Pin   = GPIO_Pin_6 | GPIO_Pin_7;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);

    TIM_TimeBaseInitTypeDef TIM_TimeBaseInitStructure;
    TIM_TimeBaseInitStructure.TIM_ClockDivision = TIM_CKD_DIV1;
    TIM_TimeBaseInitStructure.TIM_CounterMode   = TIM_CounterMode_Up;
    TIM_TimeBaseInitStructure.TIM_Prescaler     = 0;      /* ★ 编码器信号不预分频 */
    TIM_TimeBaseInitStructure.TIM_Period        = 65535;
    TIM_TimeBaseInitStructure.TIM_RepetitionCounter = 0;
    TIM_TimeBaseInit(TIM3, &TIM_TimeBaseInitStructure);

    /* ★ 这一句把 CH1/CH2 变成编码器输入，两个通道被“吃掉” */
    TIM_EncoderInterfaceConfig(TIM3, TIM_EncoderMode_TI12,
                               TIM_ICPolarity_Rising, TIM_ICPolarity_Rising);

    TIM_SetCounter(TIM3, 0);
    TIM_Cmd(TIM3, ENABLE);                                /* ★ 编码器模式不需要任何中断 */
}

/* ★ 读完就清零：永远只在小范围计数，彻底避开 16 位 CNT 回绕 */
int16_t Encoder_Get(void)
{
    int16_t temp = (int16_t)TIM_GetCounter(TIM3);
    TIM_SetCounter(TIM3, 0);
    return temp;
}

/* 增量 → rad/s（★ 注意「输出轴」要除减速比，这里 CNT_PER_REV 已含） */
float Encoder_ToRadPerSec(int16_t cnt, float dt)
{
    return (cnt / ENCODER_CNT_PER_REV) * 2.0f * 3.14159265f / dt;
}
```

> ⚠️ **一个定时器只能带一路编码器**（`TIM_EncoderInterfaceConfig` 会占用 CH1+CH2）。
> **C8T6 只有 TIM1~TIM4 → 最多 2 路编码器。**

## 5 协议帧（长度驱动状态机 + 智能重同步）

```c
/* ---------- Frame.h ---------- */
#define FRAME_HEAD        0xA5
#define FRAME_TAIL        0x0D
#define FRAME_MAX_DATA    16          /* ★ 缓冲区要留余量 */

void Frame_FeedByte(uint8_t byte);    /* ★ 每次只喂一个字节 */
void Frame_Send(const uint8_t *data, uint8_t len);

/* ---------- Frame.c ---------- */
typedef enum { ST_HEAD = 0, ST_LEN, ST_DATA, ST_SUM, ST_TAIL } FrameState_t;

static FrameState_t s_state = ST_HEAD;
static uint8_t s_len, s_idx, s_sum, s_data[FRAME_MAX_DATA];
static volatile uint8_t s_ready;

void Frame_FeedByte(uint8_t byte)
{
    switch (s_state)
    {
    case ST_HEAD:
        if (byte == FRAME_HEAD) { s_state = ST_LEN; }
        /* ★ 不是帧头就继续等，不做任何“清空” */
        break;

    case ST_LEN:
        s_len = byte;
        s_idx = 0;
        s_sum = byte;                                        /* 校验和从 len 开始累加 */
        /* ★ 长度超上限直接判非法，防止缓冲区溢出 */
        s_state = (s_len > FRAME_MAX_DATA) ? ST_HEAD : ST_DATA;
        break;

    case ST_DATA:
        s_data[s_idx++] = byte;                              /* ★ 不检查内容 */
        s_sum += byte;
        if (s_idx >= s_len) { s_state = ST_SUM; }
        break;

    case ST_SUM:
        /* ★ 校验失败 → 智能重同步（而不是清空缓冲区） */
        s_state = (byte == (s_sum & 0xFF)) ? ST_TAIL
               : ((byte == FRAME_HEAD) ? ST_LEN : ST_HEAD);
        break;

    case ST_TAIL:
        if (byte == FRAME_TAIL) { s_ready = 1; }             /* ★ 输出一帧 */
        s_state = ST_HEAD;
        break;
    }
}

void Frame_Send(const uint8_t *data, uint8_t len)
{
    uint8_t i, sum = len;

    Serial_SendByte(FRAME_HEAD);
    Serial_SendByte(len);
    for (i = 0; i < len; i++)
    {
        Serial_SendByte(data[i]);
        sum += data[i];
    }
    Serial_SendByte(sum & 0xFF);
    Serial_SendByte(FRAME_TAIL);
}
```

## 6 姿态解算（互补滤波，三段式）

```c
/* ---------- Attitude.h ---------- */
#define ATT_TAU_S   1.0f        /* ★ 时间常数 τ */
#define ATT_DT_S    0.01f       /* ★ 采样周期 10ms */

typedef struct {
    float pitch, roll;          /* 融合结果（度） */
    float pitch_acc, roll_acc;  /* ★ 加速度计单独算的（必须分开存） */
    float gx_bias, gy_bias;     /* 陀螺仪零偏（°/s） */
} Attitude_t;

void Attitude_Calibrate(Attitude_t *att);                 /* 上电静止时调一次 */
void Attitude_Sample(Attitude_t *att);                    /* ★ 主循环：只做 I2C */
void Attitude_FilterUpdate(Attitude_t *att, const int16_t *gyro_raw);  /* ★ 中断：纯数学 */

/* ---------- Attitude.c ---------- */
static float comp_alpha(void)
{
    return ATT_TAU_S / (ATT_TAU_S + ATT_DT_S);            /* = 0.9901 */
}

void Attitude_Sample(Attitude_t *att)
{
    int16_t raw[7];
    float ax, ay, az;

    MPU6050_GetData(raw);                                 /* ★ 软件 I2C，只能在主循环做 */

    ax = (float)raw[0] / 16384.0f;                        /* ±2g 量程 → 16384 LSB/g */
    ay = (float)raw[1] / 16384.0f;
    az = (float)raw[2] / 16384.0f;

    /* ★ 用 atan2（全象限 + 抗过载），不要用 asin */
    att->pitch_acc = atan2f(-ax, sqrtf(ay * ay + az * az)) * 57.29578f;
    att->roll_acc  = atan2f( ay, az)                       * 57.29578f;
}

void Attitude_FilterUpdate(Attitude_t *att, const int16_t *gyro_raw)
{
    float gx = (float)gyro_raw[0] / 131.0f;               /* ±250°/s → 131 LSB/(°/s) */
    float gy = (float)gyro_raw[1] / 131.0f;
    float a  = comp_alpha();

    gx -= att->gx_bias;                                   /* ★ 先减零偏 */
    gy -= att->gy_bias;

    /* ★ 互补滤波：陀螺仪推进（快） + 加速度计纠正（慢） */
    att->pitch = a * (att->pitch + gy * ATT_DT_S) + (1.0f - a) * att->pitch_acc;
    att->roll  = a * (att->roll  + gx * ATT_DT_S) + (1.0f - a) * att->roll_acc;
}
```

## 7 PID（位置式 + 积分限幅 + 积分分离）

```c
/* ---------- PID.h ---------- */
typedef struct {
    float Kp, Ki, Kd;
    float integral;
    float last_actual;        /* ★ 对“测量值”微分，而不是对误差微分 */
    float out_min, out_max;
    float i_max;
} PID_t;

void  PID_Init(PID_t *p, float kp, float ki, float kd);
float PID_Calc(PID_t *p, float target, float actual, float dt);

/* ---------- PID.c ---------- */
void PID_Init(PID_t *p, float kp, float ki, float kd)
{
    p->Kp = kp; p->Ki = ki; p->Kd = kd;
    p->integral = 0.0f;
    p->last_actual = 0.0f;
    p->out_min = -100.0f; p->out_max = 100.0f;
    p->i_max   = 300.0f;                                   /* ★ 积分限幅 */
}

float PID_Calc(PID_t *p, float target, float actual, float dt)
{
    float err = target - actual;
    float d, out;

    /* ① 积分：★ 带限幅 + 分离（误差大时不累加，从根上防饱和） */
    if (fabsf(err) < 2.0f)
    {
        p->integral += err;
        if (p->integral >  p->i_max) { p->integral =  p->i_max; }
        if (p->integral < -p->i_max) { p->integral = -p->i_max; }
    }

    /* ② 微分：对测量值微分，避免目标突变时的“微分冲击” */
    d = (actual - p->last_actual) / dt;
    p->last_actual = actual;

    /* ③ 输出：★ 注意 dt 对增益的影响（Ki*dt、Kd/dt） */
    out = p->Kp * err
        + p->Ki * p->integral * dt
        - p->Kd * d;

    if (out > p->out_max) { out = p->out_max; }             /* ★ 输出限幅 */
    if (out < p->out_min) { out = p->out_min; }

    return out;
}
```

> ⚠️ **`dt` 必须来自固定周期的定时中断** —— 主循环里周期不定，`Ki*dt` / `Kd/dt` 就失去意义。

## 8 调试三件套（一定要有）

```c
/* ① 自写浮点打印（★ MicroLIB 不支持 printf 的 %f） */
void Uart_PrintFloat(float value, uint8_t dec_len)
{
    int32_t int_part, dec_part;
    uint32_t scale = 1, i;

    for (i = 0; i < dec_len; i++) { scale *= 10; }

    if (value < 0.0f) { Serial_SendByte('-'); value = -value; }

    int_part = (int32_t)value;
    dec_part = (int32_t)((value - (float)int_part) * (float)scale + 0.5f);   /* 四舍五入 */

    Uart_PrintInt(int_part);
    Serial_SendByte('.');

    /* ★ 补前导零：否则 0.05 会打成 0.5，看起来像数值错了 10 倍 */
    for (i = scale / 10; i > 1; i /= 10)
    {
        if (dec_part < (int32_t)i) { Serial_SendByte('0'); }
    }
    Uart_PrintInt(dec_part);
}

/* ② VOFA+ justfloat：帧尾 0x00 0x00 0x80 0x7F */
void Vofa_SendFloats(const float *data, uint8_t count)
{
    uint8_t i;
    for (i = 0; i < count; i++)
    {
        Serial_SendArray((uint8_t *)&data[i], 4);      /* 小端 4 字节 */
    }
    Serial_SendByte(0x00); Serial_SendByte(0x00);
    Serial_SendByte(0x80); Serial_SendByte(0x7F);
}

/* ③ I2C 地址扫描（★ 调试 I2C 的万能第一步，10 秒知道总线上有谁） */
void I2C_Scan(void)
{
    uint8_t addr;
    for (addr = 0x08; addr < 0x78; addr++)
    {
        MyI2C_Start();
        MyI2C_SendByte(addr << 1);                     /* ★ 7 位地址左移一位 + 写方向 */
        if (MyI2C_ReceiveAck() == 0)                   /* ACK = 0 表示有设备应答 */
        {
            Uart_PrintString("Found: 0x");
            Uart_PrintHex(addr);
            Serial_SendString("\r\n");
        }
        MyI2C_Stop();
    }
}
```

## 9 五个「不会报错但会出问题」的错误（★ 每个都值得背）

| # | 错误 | 后果 | 正确做法 |
|:--:|---|---|---|
| 1 | 中断与主循环共享变量**没加 `volatile`** | `-O0` 能跑、`-O2` 死循环 | 全部加 `volatile` |
| 2 | **`float` 共享变量不加保护** | 读到「半新半旧」的值，显示乱码 | 双缓冲 / 复制到局部变量 |
| 3 | **换算分母与量程不对应**（如 ±2g 用了 8192） | 角度整体差倍数，**不报错** | 注释写清量程 ↔ 分母的对应 |
| 4 | **两端缩放宏不一致** | 速度差 10 倍 → **一上电窜出去** | **共享同一个头文件** |
| 5 | **`int8_t` 存超范围值** | 符号翻转 → 全速反向 | 用 `uint8_t` + 显式限幅 |
